## 三、模拟引擎主循环（v2.0）

```python
class SimulationEngine:
    def run(self) -> SimulationResult:
        t = 0.0; step = 0

        while t < self.config.T_max and not self.state.fracture_status.is_terminal:
            dt = self._compute_step_size()

            # Step 1: 更新外生变量
            self.international.update_exogenous(t)

            # Step 2: 计算部门产出
            for sector in self.sectors.values():
                sector.compute_production(self.state, dt)

            # Step 3: 劳动力市场出清
            self.labor_market.step(self.state, dt)

            # Step 4: 资本积累与投资分配
            self.accumulation.step(self.state, dt)

            # Step 5: 数字化与数据资产更新
            self.digitalization.step(self.state, dt)
            self.data_assets.step(self.state, dt)           # ← NEW

            # Step 6: 技术动态（含知识扩散）
            self.tech_dynamics.step(self.state, dt)
            self.tech_diffusion.step(self.state, dt)        # ← NEW

            # Step 7: 国际交互结算
            self.international.step(self.state, dt)

            # Step 8: SFC闭合检查 ← NEW
            self.sfc_closure.check_and_close(self.state)

            # Step 9: 异质性追踪 ← NEW
            self.digital_dependency.update(self.state)
            self.deind_diagnostic.update(self.state)
            self.households.update(self.state, dt)

            # Step 10: 福利与代价更新
            self.welfare.step(self.state, dt)
            self.cost_tracker.step(self.state, dt)

            # Step 11: EWS计算 ← NEW（在断裂检测之前！）
            self.ews_monitor.compute_all(self.state)

            # Step 12: 断裂检测
            fracture_events = self.fracture_monitor.check(self.state)
            for event in fracture_events:
                self._handle_fracture(event)

            # Step 13: 记录状态
            self.recorder.record(self.state)

            t += dt; step += 1

        return SimulationResult(...)
```

## 四、EWS监测系统（新增）

```python
class EarlyWarningMonitor:
    """
    在断裂检测之前运行。
    基于滑动窗口计算多项统计量，检测系统是否正在逼近临界点。
    """

    def __init__(self, window_size: int = 40, trend_window: int = 20):
        self.window_size = window_size
        self.trend_window = trend_window
        self.history: Dict[str, List[float]] = {}

    def compute_all(self, state: SystemState) -> Dict[str, EWSResult]:
        results = {}
        monitored_vars = {
            'K_i_ratio': state.sectors['i'].K / state.initial_K_i,
            'tech_gap': state.tech_gap,
            'tech_import_share': state.external_dependency,
            'unemployment': state.households.unemployed.population / state.total_L,
            'welfare': state.welfare_index,
        }
        for name, value in monitored_vars.items():
            self.history.setdefault(name, []).append(value)
            if len(self.history[name]) >= self.window_size:
                results[name] = self._compute_ews(name)
        return results

    def _compute_ews(self, var_name: str) -> EWSResult:
        series = np.array(self.history[var_name][-self.window_size:])

        # 滚动方差趋势
        variances = [np.var(series[i:i+10]) for i in range(0, len(series)-10, 5)]
        var_trend = kendalltau(range(len(variances)), variances).statistic

        # 滚动AR(1)趋势
        ar1s = [np.corrcoef(series[i:i+9], series[i+1:i+10])[0,1]
                for i in range(0, len(series)-10, 5)]
        ar1s = [a for a in ar1s if not np.isnan(a)]
        ar1_trend = kendalltau(range(len(ar1s)), ar1s).statistic if ar1s else 0

        # 偏度
        skew = stats.skew(series[-20:])

        # 闪烁（双模态检测）
        _, dip_pvalue = diptest(series[-30:])

        # 综合指数
        composite = (0.25 * max(0, var_trend)
                   + 0.25 * max(0, ar1_trend)
                   + 0.20 * abs(skew) / 2.0
                   + 0.15 * (1.0 - dip_pvalue)
                   + 0.15 * self._es_proximity())

        status = ('critical' if composite >= 0.7
                  else 'warning' if composite >= 0.4
                  else 'normal')

        return EWSResult(var_name, var_trend, ar1_trend, skew,
                         dip_pvalue, self._es_proximity(), composite, status)
```

## 六、关键技术风险与缓解（v2.0更新）

| 风险 | 影响 | 缓解措施 |
|------|------|---------|
| 数值不稳定 | 模拟崩溃或失真 | 自适应步长 + 隐式求解器备选 + 变量裁剪 |
| SFC闭合失败 | 会计恒等式不成立 | 未计划库存吸收残差 + 记录残差异常预警 |
| EWS误报率高 | 错误预警降低可信度 | 滞后阈值 + 多指标交叉验证 + 回溯验证 |
| 参数空间维度爆炸 | 无法充分探索 | Sobol敏感性分析→降维 + 分阶段探索 |
| 新增模块增加计算量 | 批量实验时间过长 | Numba JIT + 粗粒度并行 + 选择性关闭EWS |
| 异质性参数难校准 | 家庭群组参数无经验参照 | 宽范围采样 + 定性一致性检验 |
