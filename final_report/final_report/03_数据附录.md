## 五、数据加载示例

```python
import json, pickle

# 加载批量实验结果
with open('data/phase1_20260726_114855_results.json', 'r', encoding='utf-8') as f:
    phase1 = json.load(f)

# 提取存活矩阵
for s in phase1['summaries']:
    print(f"{s['profile']}×{s['regime']}×{s['scenario']}: {'存活' if s['survived'] else f'死亡@{s[\"survival_years\"]:.1f}y'}")

# 加载详细时间序列
with open('data/01_single_detailed_run_20260726_130707/full_results.pkl', 'rb') as f:
    df = pickle.load(f)['results'][0]['dataframe']
    
# 提取关键变量时间序列
t = df.iloc[:, 0]       # 时间
welfare = df.iloc[:, 67] # 福利
gdp = df.iloc[:, 61]     # GDP
K_i = df.iloc[:, 2]      # 工业资本
L_i = df.iloc[:, 14]     # 工业就业
DL_avg = (df.iloc[:, 21] + df.iloc[:, 22]) / 2  # 平均数字化水平

print(f"T={t.iloc[-1]:.1f}, GDP={gdp.iloc[-1]:.1f}, welfare={welfare.iloc[-1]:.3f}")
```
